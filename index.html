function parseSheetRows(rows) {
if (!rows || rows.length < 2) return;
const headers = rows[0];
availableDates = [];
for (let c = 2; c < headers.length; c++) {
let h = String(headers[c]).trim();
if (h) {
let normalized = parseAnyDate(h);
if (normalized) {
let parsedObj = parseDateDetails(normalized);
availableDates.push({ colIndex: c, dateStr: normalized, dayNum: parsedObj.dayNum, monthName: parsedObj.monthName, dayName: parsedObj.dayName });
}
}
}
for (let r = 1; r < rows.length; r++) {
let row = rows[r];
if (!row || row.length < 2) continue;
let rowNameRaw = row[1] ? String(row[1]).trim().toUpperCase() : "";
if (rowNameRaw.includes("ESCALANTE") || rowNameRaw.includes("DARIO")) {
availableDates.forEach(item => {
let val = row[item.colIndex] !== undefined ? String(row[item.colIndex]).trim() : "";
adminSchedule[item.dateStr] = val;
});
} else {
let supKey = "";
if (rowNameRaw.includes("QUIROGA")) supKey = "QUIROGA";
else if (rowNameRaw.includes("MARELLI")) supKey = "MARELLI";
else if (rowNameRaw.includes("CERVELLERA")) supKey = "CERVELLERA";
else if (rowNameRaw.includes("SALDIVIA")) supKey = "SALDIVIA";
if (supKey) {
availableDates.forEach(item => {
let val = row[item.colIndex] !== undefined ? String(row[item.colIndex]).trim() : "";
if (!rawSchedule[item.dateStr]) {
rawSchedule[item.dateStr] = {};
}
rawSchedule[item.dateStr][supKey] = val;
});
}
}
}
}
function parseDateDetails(dateStr) {
let p = dateStr.split('-');
let year = p[0], month = p[1], day = p[2];
let dObj = new Date(year, month - 1, day);
const daysNames = ['DOMINGO', 'LUNES', 'MARTES', 'MIÉRCOLES', 'JUEVES', 'VIERNES', 'SÁBADO'];
const months = { '01': 'ENE', '02': 'FEB', '03': 'MAR', '04': 'ABR', '05': 'MAY', '06': 'JUN', '07': 'JUL', '08': 'AGO', '09': 'SEP', '10': 'OCT', '11': 'NOV', '12': 'DIC' };
return { dayNum: parseInt(day), monthName: months[month] || month, dayName: daysNames[dObj.getDay()] };
}
function parseAnyDate(str) {
str = str.trim();
if (str.includes('T') || str.includes('Z')) {
return str.split('T')[0];
}
if (/^\d{4}[-/]\d{1,2}[-/]\d{1,2}/.test(str)) {
let parts = str.split(/[-/]/);
return `${parts[0]}-${parts[1].padStart(2,'0')}-${parts[2].padStart(2,'0')}`;
}
let lower = str.toLowerCase();
const months = { 'ene': '01', 'feb': '02', 'mar': '03', 'abr': '04', 'may': '05', 'jun': '06', 'jul': '07', 'ago': '08', 'sep': '09', 'oct': '10', 'nov': '11', 'dic': '12' };
let parts = lower.split(/[-/]/);
if (parts.length >= 2) {
let day = parts[0].padStart(2, '0');
let mKey = parts[1].substring(0, 3);
let month = months[mKey];
let year = parts[2] ? (parts[2].length === 2 ? '20' + parts[2] : parts[2]) : '2026';
if (month) return `${year}-${month}-${day}`;
}
let d = new Date(str);
if (!isNaN(d.getTime())) {
return formatDateKey(d);
}
return null;
}
function formatDateKey(d) {
const year = d.getFullYear();
const month = String(d.getMonth() + 1).padStart(2, '0');
const day = String(d.getDate()).padStart(2, '0');
return `${year}-${month}-${day}`;
}

function getDaysToDisplay() {
if (isSupervisorMode) {
let d = new Date(baseWeekDate);
let day = d.getDay();
let diffToMonday = d.getDate() - day + (day === 0 ? -6 : 1);
let monday = new Date(d.setDate(diffToMonday));
let weekArr = [];
for (let i = 0; i < 7; i++) {
let nextDay = new Date(monday);
nextDay.setDate(monday.getDate() + i);
weekArr.push(formatDateKey(nextDay));
}
return weekArr;
} else {
let currentItem = availableDates[currentIndex];
if (!currentItem) return [];
let targetDateStr = currentItem.dateStr;
let p = targetDateStr.split('-');
let targetDate = new Date(p[0], p[1] - 1, p[2]);
let dayOfWeek = targetDate.getDay();
let hasTodayGuard = rawSchedule[targetDateStr] && Object.values(rawSchedule[targetDateStr]).includes("G");
if (dayOfWeek >= 1 && dayOfWeek <= 5 && !hasTodayGuard) {
return [targetDateStr];
}
let backwardDays = [];
for (let i = 1; i <= 3; i++) {
let prevD = new Date(targetDate);
prevD.setDate(targetDate.getDate() - i);
let pKey = formatDateKey(prevD);
let isWeekend = (prevD.getDay() === 0 || prevD.getDay() === 6);
let hasG = rawSchedule[pKey] && Object.values(rawSchedule[pKey]).includes("G");
if (isWeekend || hasG) {
backwardDays.unshift(pKey);
} else {
break;
}
}
let forwardDays = [];
for (let i = 1; i <= 3; i++) {
let nextD = new Date(targetDate);
nextD.setDate(targetDate.getDate() + i);
let nKey = formatDateKey(nextD);
let isWeekend = (nextD.getDay() === 0 || nextD.getDay() === 6);
let hasG = rawSchedule[nKey] && Object.values(rawSchedule[nKey]).includes("G");
if (isWeekend || hasG) {
forwardDays.push(nKey);
} else {
break;
}
}
let combined = [...backwardDays, targetDateStr, ...forwardDays];
let uniqueDays = [...new Set(combined)].filter(dStr => availableDates.some(d => d.dateStr === dStr));
if (uniqueDays.length >= 2 || hasTodayGuard) {
return uniqueDays;
}
return [targetDateStr];
}
}
